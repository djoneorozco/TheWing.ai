<script>
(() => {
 'use strict';
 let queued = window.PCSU_SNAPSHOT_BRIEF_REQUEST || null;
 let handleRequest, handleReset;
 window.addEventListener('pcsu:snapshot-brief-reset', () => {
  queued = null;
  if (handleReset) handleReset();
 });
 window.addEventListener('pcsu:snapshot-brief-request', event => {
  queued = event.detail;
  if (handleRequest) handleRequest(queued);
 });
 function init() {
  const card = document.getElementById('pcsu-brief-card');
  if (!card || card.dataset.mounted) return;
  card.dataset.mounted = 'true';
  const layout = document.getElementById('pcs1-map-layout');
  const host = document.getElementById('pcs1-map-host');
  if (layout && host && layout.contains(host)) {
   if (!host.contains(card)) host.appendChild(card);
   layout.classList.remove('pcsu-demo-layout');
   layout.classList.add('pcsu-brief-layout');
  }
  const get = id => document.getElementById('pcsu-brief-' + id);
  const frame=get('frame'),cover=get('cover'),status=get('status'),live=get('live');
  const selection=get('selection'),open=get('open'),retry=get('retry');
  const appURL = new URL(card.dataset.appUrl);
  const compactView = card.dataset.view !== 'full';
  // Record the actual original cover box before changing its ready-state layout.
  // The child fills this footprint; its resize messages cannot lengthen the card.
  if (compactView) {
   card.style.height=(card.getBoundingClientRect().height || 759)+'px';
   card.dataset.fixed='true';
  }
  let requestId='', timer, lastInput=null;
  function setState(state, message) {
   card.dataset.state=state;
   card.setAttribute('aria-busy', String(state === 'loading'));
   status.textContent=message; live.textContent=message;
   cover.hidden=state === 'ready';
   card.querySelector('.brief-head').hidden=compactView && state === 'ready';
   frame.setAttribute('aria-hidden',String(state !== 'ready'));
   frame.tabIndex=state === 'ready' ? 0 : -1;
   retry.hidden=state !== 'error' || !lastInput;
   open.hidden=state !== 'ready';
  }
  function resize(height) {
   if (compactView) return;
   const n=Number(height);
   if (Number.isFinite(n) && n > 0) card.style.setProperty('--brief-height', Math.max(480,Math.min(2400,Math.ceil(n)))+'px');
  }
  function show(input) {
   const rank=String(input?.rank || '').toUpperCase().replace('-','');
   const yos=Number(input?.yos),family=Number(input?.family);
   const base=String(input?.base || ''),key=String(input?.cityKey || '');
   if (!/^[EO]\d{1,2}$/.test(rank) || input?.yos == null || !Number.isFinite(yos) || yos<0 || !Number.isInteger(family) || family<1 || !base || !/^[A-Za-z0-9.-]+$/.test(key)) return;
   clearTimeout(timer);
   lastInput={rank,yos,family,base,cityKey:key,newZip:String(input?.newZip || '')};
   requestId=crypto.randomUUID ? crypto.randomUUID() : Date.now()+'-'+Math.random().toString(36).slice(2);
   const url=new URL(appURL);
   const values={...lastInput,hasDependents:family>=2,baseFile:key+'.json',
    newBase:base,newCityKey:key,newFileKey:key,type:'ad',view:card.dataset.view || 'brief'};
   Object.entries(values).forEach(([key,value])=>url.searchParams.set(key,String(value)));
   const snapshot=new URL('/PCS-Snapshot/index.html',appURL.origin);
   const handoff={rank,yos,family,hasDependents:family>=2,newBase:base,newCityKey:key,newFileKey:key,newZip:lastInput.newZip,type:'ad'};
   Object.entries(handoff).forEach(([key,value])=>snapshot.searchParams.set(key,String(value)));
   open.href=snapshot.href;
   url.searchParams.set('parentOrigin',location.origin);
   url.searchParams.set('requestId',requestId);
   selection.textContent=base+' · '+rank.replace(/^([EO])/,'$1-')+' · '+yos+' years · '+family+(family===1?' person':' people');
   frame.title=base+' — PCS Snapshot Brief';
   card.style.removeProperty('--brief-height');
   setState('loading',card.dataset.view==='pay'?'Preparing your monthly compensation…':'Preparing your pay, destination, and housing brief…');
   frame.src=url.href;
   timer=setTimeout(()=>{requestId='';setState('error','Your brief has not confirmed it is ready. Please retry shortly.');},30000);
  }
  window.addEventListener('message',event=>{
   if (event.origin!==appURL.origin || event.source!==frame.contentWindow) return;
   const data=event.data;
   if (!data || data.source!=='pcsu-snapshot-brief' || !requestId || data.requestId!==requestId) return;
   if (data.type==='pcsu:brief-status' && card.dataset.state==='loading') {
    if (data.state==='ready') {clearTimeout(timer);resize(data.height);setState('ready','Your PCS Snapshot Brief is ready.');}
    else if (data.state==='error') {clearTimeout(timer);requestId='';setState('error','Your brief could not load. Please retry.');}
   } else if (data.type==='pcsu:brief-resize' && card.dataset.state==='ready') resize(data.height);
  });
  // Navigation/load events cannot prove API success; only a matching child message can.
  handleReset=()=>{
   clearTimeout(timer);requestId='';lastInput=null;
   frame.removeAttribute('src');open.removeAttribute('href');
   selection.textContent='Your next base, at a glance';
   card.style.removeProperty('--brief-height');
   setState('idle','Select Analyze to view your updated brief.');
  };
  retry.addEventListener('click',()=>{if(lastInput)show(lastInput);});
  setState('idle','Enter your rank, years of service, family size, and destination. Then select Analyze.');
  handleRequest=show;
  if (queued) show(queued);
 }
 if (document.readyState==='loading') document.addEventListener('DOMContentLoaded',init,{once:true});
 else init();
})();
</script>
